    name: cat.name,
    slug: cat.slug,
    parentId: cat.parentId,
    label: cat.name,
    imageUrl: cat.imageUrl ?? null,
  }));

    .filter((cat) => cat.products && cat.products.length > 0)

        id: cat.id,
        name: cat.name,
        slug: cat.slug,
        parentId: cat.parentId,
        label: cat.name,
        imageUrl: cat.imageUrl ?? null,
      },
      products: cat.products.slice(0, 4).map((p) => {
        const defaultVariant = p.variants?.[0];
        const inStock = Boolean(p.variants?.some((v) => v.stock > 0));

        return {
          id: p.id,
          name: p.name,
          slug: p.slug,
          shortDescription: p.description ?? "",
          price: defaultVariant?.price ? String(defaultVariant.price) : "0",
          regularPrice: defaultVariant?.compareAtPrice ? String(defaultVariant.compareAtPrice) : null,
          image: p.images?.[0]?.url || "/branding/bimlay-logo.png",
          imageAlt: p.images?.[0]?.alt || p.name,
          inStock,
          hasMultipleVariants: (p.variants?.length ?? 0) > 1,
        };
      }),
    }));

  const allProducts = featuredProducts.map((p) => {
    const defaultVariant = p.variants?.[0];
    const inStock = Boolean(p.variants?.some((v) => v.stock > 0));

    return {
      id: p.id,
      name: p.name,
      slug: p.slug,

"use server";

import { createHash } from "node:crypto";
import { redirect } from "next/navigation";
import { db } from "@/lib/db";

import {
  hashPassword,
  verifyPassword,
} from "@/lib/password";
import {
  createCustomerSession,
  deleteCustomerSession,
  requireCustomer,
} from "@/lib/customer-auth";

import { revalidatePath } from "next/cache";

export type CustomerAuthState = {
  error: string;
};

function text(formData: FormData, key: string) {
  const value = formData.get(key);
  return typeof value === "string" ? value.trim() : "";
}

function passwordValue(formData: FormData, key: string) {
  const value = formData.get(key);
  return typeof value === "string" ? value : "";
}

function validEmail(email: string) {
  return (
    email.length <= 254 &&
    /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
  );
}

async function allowAttempt(
  email: string,
  purpose: "login" | "register",
) {
  const windowMs = 15 * 60 * 1000;
  const windowNumber = Math.floor(Date.now() / windowMs);

  const id = createHash("sha256")
    .update(`customer:${purpose}:${email}:${windowNumber}`)
    .digest("hex");

  const limit = await db.customerAuthLimit.upsert({
    where: { id },
    create: {
      id,
      attempts: 1,
      expiresAt: new Date((windowNumber + 1) * windowMs),
    },
    update: {
      attempts: { increment: 1 },
    },
  });

  return limit.attempts <= (purpose === "login" ? 8 : 5);
}

export async function registerCustomer(
  _previousState: CustomerAuthState,
  formData: FormData,
): Promise<CustomerAuthState> {
  const name = text(formData, "name");
  const email = text(formData, "email").toLowerCase();
  const password = passwordValue(formData, "password");
  const confirmation = passwordValue(
    formData,
    "confirmPassword",
  );

  if (name.length < 2 || name.length > 100) {
    return { error: "Enter a name containing 2 to 100 characters." };
  }

  if (!validEmail(email)) {
    return { error: "Enter a valid email address." };
  }

  if (password.length < 12 || password.length > 128) {
    return { error: "Password must contain 12 to 128 characters." };
  }

  if (password !== confirmation) {
    return { error: "Passwords do not match." };
  }

  try {
    if (!(await allowAttempt(email, "register"))) {
      return {
        error: "Too many attempts. Try again after 15 minutes.",
      };
    }

    const existing = await db.customer.findUnique({
      where: { email },
      select: { id: true },
    });

    if (existing) {
      return {
        error: "Registration is unavailable for this email. Try signing in.",
      };
    }

    const passwordHash = await hashPassword(password);

    const customer = await db.customer.create({
      data: {
        name,
        email,
        passwordHash,
      },
      select: { id: true },
    });

    await createCustomerSession(customer.id);
  } catch {
    return {
      error: "Could not finish registration. If you already registered, try signing in.",
    };
  }

  redirect("/account");
}

const DUMMY_HASH = [
  "scrypt",
  "131072",
  "8",
  "1",
  "0".repeat(32),
  "0".repeat(128),
].join("$");

export async function loginCustomer(
  _previousState: CustomerAuthState,
  formData: FormData,
): Promise<CustomerAuthState> {
  const email = text(formData, "email").toLowerCase();
  const password = passwordValue(formData, "password");

  if (
    !validEmail(email) ||
    password.length < 1 ||
    password.length > 128
  ) {
    return { error: "Email or password is incorrect." };
  }

  try {
    if (!(await allowAttempt(email, "login"))) {
      return {
        error: "Too many attempts. Try again after 15 minutes.",
      };
    }

    const customer = await db.customer.findUnique({
      where: { email },
      select: {
        id: true,
        passwordHash: true,
        isActive: true,
      },
    });

    const valid = await verifyPassword(
      password,
      customer?.passwordHash ?? DUMMY_HASH,
    );

    if (!customer || !customer.isActive || !valid) {
      return { error: "Email or password is incorrect." };
    }

    await createCustomerSession(customer.id);

    await db.customerAuthLimit.deleteMany({
      where: {
        expiresAt: { lte: new Date() },
      },
    });
  } catch {
    return {
      error: "Sign in is temporarily unavailable. Please try again.",
    };
  }

  redirect("/account");
}

export async function logoutCustomer(): Promise<void> {
  await deleteCustomerSession();
  redirect("/account/login");
}

export async function updateCustomerProfile(
  _previousState: CustomerProfileState,

  const name = text(formData, "name");

  if (name.length < 2 || name.length > 100) {

        success: "",
      };
    }
  } catch {
    return {
      error: "Could not save your profile. Please try again.",
      success: "",
    };
  }

  revalidatePath("/account");
  revalidatePath("/account/profile");

  return {
    error: "",
    success: "Your profile has been updated.",
  };
}

export async function changeCustomerPassword(
  _previousState: CustomerAuthState,
  formData: FormData,
): Promise<CustomerAuthState> {
  const customer = await requireCustomer();

  const currentPassword = passwordValue(
    formData,
    "currentPassword",
  );

  const newPassword = passwordValue(
    formData,
    "newPassword",
  );

  const confirmation = passwordValue(
    formData,
    "confirmPassword",
  );

  if (
    currentPassword.length < 1 ||
    currentPassword.length > 128
  ) {
    return { error: "Enter your current password." };
  }

  if (newPassword.length < 12 || newPassword.length > 128) {
    return {
      error: "New password must contain 12 to 128 characters.",
    };
  }

  if (newPassword !== confirmation) {
    return { error: "New passwords do not match." };
  }

  if (newPassword === currentPassword) {
    return {
      error: "Choose a password different from your current password.",
    };
  }

  try {
    const windowMs = 15 * 60 * 1000;
    const windowNumber = Math.floor(Date.now() / windowMs);

    const limitId = createHash("sha256")
      .update(`customer:password:${customer.id}:${windowNumber}`)
      .digest("hex");

    const limit = await db.customerAuthLimit.upsert({
      where: { id: limitId },
      create: {
        id: limitId,
        attempts: 1,
        expiresAt: new Date((windowNumber + 1) * windowMs),
      },
      update: {
        attempts: { increment: 1 },
      },
    });

    if (limit.attempts > 5) {
      return {
        error: "Too many attempts. Try again after 15 minutes.",
      };
    }

    const account = await db.customer.findUnique({
      where: { id: customer.id },
      select: {
        passwordHash: true,
        isActive: true,
      },
    });

    if (!account?.isActive) {
      return {
        error: "Your account is unavailable. Please sign in again.",
      };
    }

    const valid = await verifyPassword(
      currentPassword,
      account.passwordHash,
    );

    if (!valid) {
      return { error: "Your current password is incorrect." };
    }

    const newHash = await hashPassword(newPassword);

    await db.$transaction(async (tx) => {
      const changed = await tx.customer.updateMany({
        where: {
          id: customer.id,
          isActive: true,
          passwordHash: account.passwordHash,
        },
        data: {
          passwordHash: newHash,
        },
      });

      if (changed.count !== 1) {
        throw new Error("Account changed during password update.");
      }

      await tx.customerSession.deleteMany({
        where: { customerId: customer.id },
      });
    });
  } catch {
    return {
      error: "Could not change your password. Please sign in again or retry.",
    };
  }

  await deleteCustomerSession();
  redirect("/account/login");
}

type Props = {
  searchParams: Promise<Record<string, string | string[] | undefined>>;

async function AccountContent({ searchParams }: Props) {
  await connection();

  const customer = await requireCustomer();
  const params = await searchParams;

  const rawPage = Number(typeof params.page === "string" ? params.page : "1");

  const requestedPage =
    Number.isSafeInteger(rawPage) && rawPage > 0 ? rawPage : 1;

  const [categories, total] = await Promise.all([
    getStoreCategories(),
    db.order.count({ where }),
  ]);

  const page = Math.min(requestedPage, totalPages);

  const orders = await db.order.findMany({
    where,
    skip: (page - 1) * pageSize,
    take: pageSize,
    orderBy: [{ createdAt: "desc" }, { id: "desc" }],
    select: {
      id: true,
      orderNumber: true,
      status: true,
      paymentStatus: true,
      total: true,
      createdAt: true,
    },
  });

  return (
    <Storefront mode="account" categories={categories} products={[]}>
      <section className="shell customer-account">
        <header className="customer-account-heading">
          <div>
            <p className="eyebrow">Your personal space</p>
            <h1>Hello, {customer.name}.</h1>
            <p>{customer.email}</p>
          </div>

          <div className="customer-account-actions">
            <Link href="/account/profile" className="product-small-button">
              Account settings
            </Link>

            <form action={logoutCustomer}>
              <button type="submit" className="product-small-button">
                Sign out
              </button>
            </form>
          </div>
        </header>

        <section className="customer-orders">
          <div className="section-heading">
            <h2>Your orders</h2>
            <span className="category-help">{total} total</span>
          </div>

          {orders.length === 0 ? (
            <div className="store-empty">
              <p>No orders yet.</p>
              <p>Orders placed while signed in will appear here.</p>
              <Link href="/shop" className="primary-button">
                Explore the collection
              </Link>
            </div>
          ) : (
            <div className="customer-order-list">
              {orders.map((order) => (
                <article className="customer-order-card" key={order.id}>
                  <div>
                    <Link
                      href={`/account/orders/${order.id}`}
                      className="customer-order-link"
                    >
                      {order.orderNumber}
                    </Link>
                    <p>
                      {new Intl.DateTimeFormat("en-GB", {
                        dateStyle: "medium",
                        timeZone: "Asia/Dhaka",

                  </div>

                  <div>

              {page > 1 && (

              )}

              <span>
                Page {page} of {totalPages}
              </span>

              {page < totalPages && (
                <Link href={`/account?page=${page + 1}`}>Next</Link>
              )}
            </nav>
          )}
        </section>
      </section>
    </Storefront>
  );
}

"use server";

import { db } from "@/lib/db";
import { Prisma } from "@/generated/prisma/client";

type CartInput = {
  variantId: string;
  quantity: number;
};

export type CartLine = {
  variantId: string;
  quantity: number;
  name: string;
  slug: string | null;
  image: string | null;
  imageAlt: string;
  option: string;
  stock: number;
  unitPrice: string | null;
  lineTotal: string | null;
  issue: string;
};

export type CartQuoteResult =
  | {
      ok: true;
      lines: CartLine[];
      subtotal: string;
      hasIssues: boolean;
    }
  | {
      ok: false;
      error: string;
    };

const visibleProduct: Prisma.ProductWhereInput = {
  status: "ACTIVE",
  images: { some: {} },
  category: {
    is: {
      isActive: true,
      OR: [
        { parentId: null },
        { parent: { is: { isActive: true } } },
      ],
    },
  },
};

function parseCartInput(input: unknown): CartInput[] | null {
  if (!Array.isArray(input) || input.length > 50) {
    return null;
  }

  const items: CartInput[] = [];
  const seen = new Set<string>();

  for (const value of input) {
    if (
      !value ||
      typeof value !== "object" ||
      typeof value.variantId !== "string" ||
      value.variantId.length < 1 ||
      value.variantId.length > 100 ||
      !Number.isInteger(value.quantity) ||
      value.quantity < 1 ||
      value.quantity > 99 ||
      seen.has(value.variantId)
    ) {
      return null;
    }

    seen.add(value.variantId);

    items.push({
      variantId: value.variantId,
      quantity: value.quantity,
    });
  }

  return items;
}

export async function validateCartItem(
  variantId: string,
  quantity: number,
): Promise<
  | { ok: true; variantId: string; quantity: number }
  | { ok: false; error: string }
> {
  if (
    typeof variantId !== "string" ||
    variantId.length < 1 ||
    variantId.length > 100 ||
    !Number.isInteger(quantity) ||
    quantity < 1 ||
    quantity > 99
  ) {
    return {
      ok: false,
      error: "Choose a quantity between 1 and 99.",
    };
  }

  try {
    const variant = await db.productVariant.findFirst({
      where: {
        id: variantId,
        isActive: true,
        product: { is: visibleProduct },
      },
      select: {
        id: true,
        stock: true,
      },
    });

    if (!variant || variant.stock < 1) {
      return {
        ok: false,
        error: "This item is currently unavailable.",
      };
    }

    if (quantity > variant.stock) {
      return {
        ok: false,
        error: `Only ${variant.stock} units are available.`,
      };
    }

    return {
      ok: true,
      variantId: variant.id,
      quantity,
    };
  } catch {
    return {
      ok: false,
      error: "Could not check availability. Please try again.",
    };
  }
}

export async function quoteCart(
  input: unknown,
): Promise<CartQuoteResult> {
  const items = parseCartInput(input);

  if (!items) {
    return {
      ok: false,
      error: "Your bag contains invalid data.",
    };
  }

  if (items.length === 0) {
    return {
      ok: true,
      lines: [],
      subtotal: "0.00",
      hasIssues: false,
    };
  }

  try {
    const variants = await db.productVariant.findMany({
      where: {
        id: {
          in: items.map((item) => item.variantId),
        },
        isActive: true,
        product: { is: visibleProduct },
      },
      select: {
        id: true,
        size: true,
        color: true,
        price: true,
        stock: true,
        product: {
          select: {
            name: true,
            slug: true,
            images: {
              orderBy: { sortOrder: "asc" },
              take: 1,
              select: {
                url: true,
                alt: true,
              },
            },
          },
        },
      },
    });

    const byId = new Map(
      variants.map((variant) => [variant.id, variant]),
    );

    let subtotal = new Prisma.Decimal(0);

    const lines: CartLine[] = items.map((item) => {
      const variant = byId.get(item.variantId);

      if (!variant) {
        return {
          ...item,
          name: "Unavailable item",
          slug: null,
          image: null,
          imageAlt: "",
          option: "",
          stock: 0,
          unitPrice: null,
          lineTotal: null,
          issue: "This item is no longer available. Please remove it.",
        };
      }

      const issue =
        variant.stock === 0
          ? "Currently out of stock."
          : item.quantity > variant.stock
            ? `Only ${variant.stock} units are available. Reduce the quantity.`
            : "";

      const lineTotal = issue
        ? null
        : variant.price.mul(item.quantity);

      if (lineTotal) {
        subtotal = subtotal.add(lineTotal);
      }

      return {
        ...item,
        name: variant.product.name,
        slug: variant.product.slug,
        image: variant.product.images[0]?.url ?? null,
        imageAlt:
          variant.product.images[0]?.alt ?? variant.product.name,
        option:
          [variant.size, variant.color]
            .filter(Boolean)
            .join(" / ") || "Standard",
        stock: variant.stock,
        unitPrice: variant.price.toFixed(2),
        lineTotal: lineTotal?.toFixed(2) ?? null,
        issue,
      };
    });

    return {
      ok: true,
      lines,
      subtotal: subtotal.toFixed(2),
      hasIssues: lines.some((line) => Boolean(line.issue)),
    };
  } catch {
    return {
      ok: false,
      error: "Could not load your bag. Please try again.",
    };
  }
}

"use server";

import { createHash, randomBytes } from "node:crypto";
import { cookies } from "next/headers";
import { revalidatePath } from "next/cache";
import { db } from "@/lib/db";
import { Prisma } from "@/generated/prisma/client";
import { getCustomer } from "@/lib/customer-auth";

  CouponValidationError,
  normalizeCouponCode,

class CheckoutError extends Error {}

type CheckoutItem = {
  variantId: string;
  quantity: number;
};

type CheckoutResult =
  | {
      ok: true;
      orderId: string;
      items: CheckoutItem[];
    }
  | {
      ok: false;
      error: string;
    };

function hash(value: string) {
  return createHash("sha256").update(value).digest("hex");
}

function stringValue(value: unknown): string {
  return typeof value === "string" ? value.trim() : "";
}

function parseItems(value: unknown): CheckoutItem[] {
  if (!Array.isArray(value) || value.length < 1 || value.length > 50) {
    throw new CheckoutError("Your bag is empty or invalid.");
  }

  const seen = new Set<string>();

  return value.map((item) => {
    if (
      !item ||
      typeof item.variantId !== "string" ||
      item.variantId.length < 1 ||
      item.variantId.length > 100 ||
      !Number.isInteger(item.quantity) ||
      item.quantity < 1 ||
      item.quantity > 99 ||
      seen.has(item.variantId)
    ) {
      throw new CheckoutError("Your bag contains invalid items.");
    }

    seen.add(item.variantId);

    return {
      variantId: item.variantId,
      quantity: item.quantity,
    };
  });
}

function normalizePhone(value: string) {
  const digits = value.replace(/[০-৯]/g, (digit) =>
    String("০১২৩৪৫৬৭৮৯".indexOf(digit)),
  );

  return digits.replace(/[\s()-]/g, "").replace(/^\+?88/, "");
}

async function grantReceiptAccess(orderId: string, secret: string) {
  const cookieStore = await cookies();

  cookieStore.set("bimlay_order_receipt", `${orderId}:${secret}`, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    path: "/order-confirmation",
    maxAge: 48 * 60 * 60,
  });
}

export async function submitCheckout(input: unknown): Promise<CheckoutResult> {
  try {
    if (!input || typeof input !== "object") {
      throw new CheckoutError("Invalid checkout request.");
    }

    const data = input as Record<string, unknown>;

    const checkoutKey = stringValue(data.checkoutKey);
    const receiptSecret = stringValue(data.receiptSecret);

    if (
      !/^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(
        checkoutKey,
      ) ||
      !/^[a-f0-9]{64}$/.test(receiptSecret)
    ) {
      throw new CheckoutError("Invalid checkout session.");
    }

    const receiptTokenHash = hash(receiptSecret);

    // A retry must return the original order.
    const previous = await db.order.findUnique({
      where: { checkoutKey },
      include: { items: true },
    });

    if (previous) {
      if (previous.receiptTokenHash !== receiptTokenHash) {
        throw new CheckoutError("Invalid checkout session.");
      }

      await grantReceiptAccess(previous.id, receiptSecret);

      return {
        ok: true,
        orderId: previous.id,
        items: previous.items.flatMap((item) =>
          item.variantId
            ? [
                {
                  variantId: item.variantId,
                  quantity: item.quantity,
                },
              ]
            : [],
        ),
      };
    }

    const customerName = stringValue(data.customerName);
    const phone = normalizePhone(stringValue(data.phone));
    const email = stringValue(data.email).toLowerCase();
    const district = stringValue(data.district);
    const address = stringValue(data.address);
    const notes = stringValue(data.notes);
    const deliveryZone = stringValue(data.deliveryZone);
    const expectedTotal = stringValue(data.expectedTotal);
    const couponInput = stringValue(data.couponCode);
    const couponCode = couponInput ? normalizeCouponCode(couponInput) : null;
    const items = parseItems(data.items);

    if (customerName.length < 2 || customerName.length > 100) {
      throw new CheckoutError("Enter your full name.");
    }

    if (!/^01\d{9}$/.test(phone)) {
      throw new CheckoutError(
        "Enter a valid 11-digit Bangladesh mobile number.",
      );
    }

    if (
      email &&
      (email.length > 254 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email))
    ) {
      throw new CheckoutError("Enter a valid email or leave it empty.");
    }

    if (district.length < 2 || district.length > 80) {
      throw new CheckoutError("Enter your district.");
    }

    if (address.length < 10 || address.length > 500) {
      throw new CheckoutError("Enter your complete delivery address.");
    }

    if (notes.length > 500) {
      throw new CheckoutError("Order notes must not exceed 500 characters.");
    }

    if (deliveryZone !== "DHAKA" && deliveryZone !== "OUTSIDE_DHAKA") {
      throw new CheckoutError("Choose a delivery area.");
    }

    if (!/^\d{1,14}\.\d{2}$/.test(expectedTotal)) {
      throw new CheckoutError("Refresh checkout and review your total.");
    }

    // Basic phone-based checkout attempt limit.
    const hour = Math.floor(Date.now() / 3600000);

    const limit = await db.checkoutLimit.upsert({
      where: {
        id: hash(`${phone}:${hour}`),
      },
      create: {
        id: hash(`${phone}:${hour}`),
        attempts: 1,
        expiresAt: new Date((hour + 1) * 3600000),
      },
      update: {
        attempts: { increment: 1 },
      },
    });

    if (limit.attempts > 8) {
      throw new CheckoutError(
        "Too many checkout attempts. Please try again later.",
      );
    }

    const customer = await getCustomer();

    for (let attempt = 0; attempt < 3; attempt += 1) {
      try {
        const order = await db.$transaction(
          async (tx) => {
            const existing = await tx.order.findUnique({
              where: { checkoutKey },
              include: { items: true },
            });

            if (existing) {
              if (existing.receiptTokenHash !== receiptTokenHash) {
                throw new CheckoutError("Invalid checkout session.");
              }

              return existing;
            }

            const settings = await tx.storeSetting.findUnique({
              where: { id: "main" },
            });

            if (!settings?.codEnabled) {
              throw new CheckoutError("Orders are temporarily unavailable.");
            }

            const variants = await tx.productVariant.findMany({
              where: {
                id: {
                  in: items.map((item) => item.variantId),
                },
                isActive: true,
                product: {
                  is: {
                    status: "ACTIVE",
                    images: { some: {} },
                    category: {
                      is: {
                        isActive: true,
                        OR: [
                          { parentId: null },
                          {
                            parent: {
                              is: { isActive: true },
                            },
                          },
                        ],
                      },
                    },
                  },
                },
              },
              select: {
                id: true,
                sku: true,
                size: true,
                color: true,
                price: true,
                stock: true,
                productId: true,
                product: {
                  select: {
                    name: true,
                    images: {
                      orderBy: { sortOrder: "asc" },
                      take: 1,
                      select: { url: true },
                    },
                  },
                },
              },
            });

            const byId = new Map(
              variants.map((variant) => [variant.id, variant]),
            );

            let subtotal = new Prisma.Decimal(0);

            const orderItems = items.map((item) => {
              const variant = byId.get(item.variantId);

              if (!variant || variant.stock < item.quantity) {
                throw new CheckoutError(
                  "An item is unavailable or its stock has changed. Review your bag.",
                );
              }

              const lineTotal = variant.price.mul(item.quantity);
              subtotal = subtotal.add(lineTotal);

              return {
                productId: variant.productId,
                variantId: variant.id,
                productName: variant.product.name,
                sku: variant.sku,
                size: variant.size,
                color: variant.color,
                imageUrl: variant.product.images[0]?.url ?? null,
                quantity: item.quantity,
                unitPrice: variant.price,
                lineTotal,
              };
            });

            const deliveryFee =
              deliveryZone === "DHAKA"
                ? settings.dhakaDeliveryCharge
                : settings.outsideDeliveryCharge;

            const coupon = couponCode
              ? await tx.coupon.findUnique({
                  where: { code: couponCode },
                })
              : null;

            if (couponCode && !coupon) {
              throw new CheckoutError("Coupon code is invalid.");
            }

            const discount = coupon
              ? calculateCouponDiscount(coupon, subtotal, new Date())
              : new Prisma.Decimal(0);

            const total = subtotal.sub(discount).add(deliveryFee);

            if (!total.eq(new Prisma.Decimal(expectedTotal))) {
              throw new CheckoutError(
                "Pricing or delivery charges changed. Refresh checkout and review the updated total before ordering.",
              );
            }

            if (coupon) {
              const claimed = await tx.coupon.updateMany({
                where: {
                  id: coupon.id,
                  isActive: true,
                  ...(coupon.usageLimit !== null
                    ? {
                        usedCount: {
                          lt: coupon.usageLimit,
                        },
                      }
                    : {}),
                },
                data: {
                  usedCount: { increment: 1 },
                },
              });

              if (claimed.count !== 1) {
                throw new CheckoutError(
                  "This coupon is no longer available. Review your total.",
                );
              }
            }

            for (const item of items) {
              const changed = await tx.productVariant.updateMany({
                where: {
                  id: item.variantId,
                  isActive: true,
                  stock: { gte: item.quantity },
                },
                data: {
                  stock: { decrement: item.quantity },
                },
              });

              if (changed.count !== 1) {
                throw new CheckoutError(
                  "Stock changed while placing your order. Review your bag.",
                );
              }
            }

            return tx.order.create({
              data: {
                orderNumber: `BL-${randomBytes(6).toString("hex").toUpperCase()}`,
                checkoutKey,
                receiptTokenHash,
                customerId: customer?.id ?? null,
                customerName,
                phone,
                email: email || null,
                district,
                address,
                notes: notes || null,
                deliveryZone,
                subtotal,
                deliveryFee,
                discount,
                couponCode,
                couponId: coupon?.id ?? null,
                total,
                items: {
                  create: orderItems,
                },
              },
              include: { items: true },
            });
          },
          {
            isolationLevel: Prisma.TransactionIsolationLevel.Serializable,
            maxWait: 5000,
            timeout: 15000,
          },
        );

        await grantReceiptAccess(order.id, receiptSecret);

        revalidatePath("/admin");
        revalidatePath("/admin/products");
        revalidatePath("/shop");
        revalidatePath("/");

        return {
          ok: true,
          orderId: order.id,
          items: order.items.flatMap((item) =>
            item.variantId
              ? [
                  {
                    variantId: item.variantId,
                    quantity: item.quantity,
                  },
                ]
              : [],
          ),
        };
      } catch (error) {
        const retryable =
          error instanceof Prisma.PrismaClientKnownRequestError &&
          (error.code === "P2034" || error.code === "P2002");

        if (retryable && attempt < 2) {
          continue;
        }

        throw error;
      }
    }

    throw new CheckoutError("Please try checkout again.");
  } catch (error) {
    return {
      ok: false,
      error:
       error instanceof CheckoutError ||
error instanceof CouponValidationError
  ? error.message
  : "Could not complete checkout. Please retry from this page."
    };
  }
}

"use server";

import { createHash } from "node:crypto";
import { redirect } from "next/navigation";
import { db } from "@/lib/db";
import { verifyPassword } from "@/lib/password";
import {
  createAdminSession,
  deleteAdminSession,
} from "@/lib/admin-auth";

export type LoginState = {
  error: string;
};

const WINDOW_MS = 15 * 60 * 1000;
const MAX_ATTEMPTS = 8;

// Nonexistent accounts still perform a password hash check.
const DUMMY_HASH = [
  "scrypt",
  "131072",
  "8",
  "1",
  "0".repeat(32),
  "0".repeat(128),
].join("$");

export async function loginAdmin(
  _previousState: LoginState,
  formData: FormData,
): Promise<LoginState> {
  const emailValue = formData.get("email");
  const passwordValue = formData.get("password");

  if (
    typeof emailValue !== "string" ||
    typeof passwordValue !== "string"
  ) {
    return { error: "Enter your email and password." };
  }

  const email = emailValue.trim().toLowerCase();
  const password = passwordValue;

  if (
    email.length > 254 ||
    !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) ||
    password.length < 1 ||
    password.length > 128
  ) {
    return { error: "Email or password is incorrect." };
  }

  try {
    const windowNumber = Math.floor(Date.now() / WINDOW_MS);

    const limitId = createHash("sha256")
      .update(`${email}:${windowNumber}`)
      .digest("hex");

    const limit = await db.adminLoginLimit.upsert({
      where: { id: limitId },
      create: {
        id: limitId,
        attempts: 1,
        expiresAt: new Date((windowNumber + 1) * WINDOW_MS),
      },
      update: {
        attempts: { increment: 1 },
      },
    });

    if (limit.attempts > MAX_ATTEMPTS) {
      return {
        error: "Too many login attempts. Try again after 15 minutes.",
      };
    }

    const admin = await db.adminUser.findUnique({
      where: { email },
      select: {
        id: true,
        passwordHash: true,
        isActive: true,
      },
    });

    const validPassword = await verifyPassword(
      password,
      admin?.passwordHash ?? DUMMY_HASH,
    );

    if (!admin || !admin.isActive || !validPassword) {
      return { error: "Email or password is incorrect." };
    }

    await db.adminLoginLimit.deleteMany({
      where: {
        expiresAt: { lte: new Date() },
      },
    });

    await createAdminSession(admin.id);
  } catch {
    return {
      error: "Login is temporarily unavailable. Please try again.",
    };
  }

  // redirect must stay outside the try/catch.
  redirect("/admin");
}

export async function logoutAdmin(): Promise<void> {
  await deleteAdminSession();
  redirect("/admin/login");
}

import { revalidatePath } from "next/cache";

  MEDIA_FILENAME_PATTERN,
  mediaImageExists,
} from "@/lib/media-storage";

export type CategoryFormState = {
  error: string;
  success: string;
};

function readText(formData: FormData, key: string): string {
  const value = formData.get(key);
  return typeof value === "string" ? value.trim() : "";
}

export async function saveCategory(
  _previousState: CategoryFormState,
  formData: FormData,
): Promise<CategoryFormState> {
  const admin = await requireAdmin();

  if (admin.role !== "OWNER") {
    return {
      error: "Only the store owner can manage categories.",
      success: "",
    };
  }

  const id = readText(formData, "id");
  const name = readText(formData, "name");
  const slug = readText(formData, "slug").toLowerCase();
  const description = readText(formData, "description");
  const imageUrl = readText(formData, "imageUrl");
  const parentId = readText(formData, "parentId") || null;
  const orderText = readText(formData, "sortOrder");
  const sortOrder = Number(orderText);
  const isActive = formData.get("isActive") === "on";

  if (name.length < 2 || name.length > 80) {
    return {
      error: "Category name must contain 2 to 80 characters.",
      success: "",
    };
  }

  if (
    slug.length < 2 ||
    slug.length > 100 ||
    !/^[a-z0-9]+(?:-[a-z0-9]+)*$/.test(slug)
  ) {
    return {
      error:
        "Use an English slug with lowercase letters, numbers and hyphens. Example: mens-shirts",
      success: "",
    };
  }

  if (description.length > 1000) {
    return {
      error: "Description must not exceed 1000 characters.",
      success: "",
    };
  }

  if (
    orderText === "" ||
    !Number.isInteger(sortOrder) ||
    sortOrder < 0 ||
    sortOrder > 9999
  ) {
    return {
      error: "Display order must be a whole number from 0 to 9999.",
      success: "",
    };
  }

  if (id.length > 100 || (parentId && parentId.length > 100)) {
    return {
      error: "Invalid category.",
      success: "",
    };
  }

  try {
    if (imageUrl) {
  const filename = imageUrl.startsWith("/media/")
    ? imageUrl.slice("/media/".length)
    : "";

  if (
    !MEDIA_FILENAME_PATTERN.test(filename) ||
    !(await mediaImageExists(imageUrl))
  ) {
    return {
      error: "Choose an uploaded category image.",
      success: "",
    };
  }
}
    const validationError = await db.$transaction(
      async (tx): Promise<string | null> => {
        if (id) {
          const existing = await tx.category.findUnique({
            where: { id },
            select: {
              id: true,
              _count: {
                select: { children: true },
              },
            },
          });

          if (!existing) {
            return "Category no longer exists. Refresh this page.";
          }

          if (parentId === id) {
            return "A category cannot be its own parent.";
          }

          if (parentId && existing._count.children > 0) {
            return "A category with subcategories must remain a main category.";
          }
        }

        if (parentId) {
          const parent = await tx.category.findUnique({
            where: { id: parentId },
            select: {
              id: true,
              parentId: true,
            },
          });

          if (!parent || parent.parentId !== null) {
            return "Choose an existing main category as the parent.";
          }
        }

        const data = {
          name,
          slug,
          description: description || null,
          imageUrl: imageUrl || null,
          parentId,
          sortOrder,
          isActive,
        };

        if (id) {
          await tx.category.update({
            where: { id },
            data,
          });
        } else {
          await tx.category.create({ data });
        }

        return null;
      },
      {
        isolationLevel: Prisma.TransactionIsolationLevel.Serializable,
      },
    );

    if (validationError) {
      return { error: validationError, success: "" };
    }
  } catch (error) {
    if (error instanceof Prisma.PrismaClientKnownRequestError) {
      if (error.code === "P2002") {
        return {
          error: "This slug is already used. Choose another slug.",
          success: "",
        };
      }

      if (error.code === "P2034") {
        return {
          error: "Another update happened at the same time. Please save again.",
          success: "",
        };
      }
    }

    return {
      error: "Could not save the category. Please try again.",
      success: "",
    };
  }

  revalidatePath("/admin/categories");
  revalidatePath("/admin");
  revalidatePath("/");
revalidatePath("/shop");

  return {
    error: "",
    success: id ? "Category updated." : "Category created.",
  };
}

"use server";

import { revalidatePath } from "next/cache";
import { Prisma } from "@/generated/prisma/client";
import { requireAdmin } from "@/lib/admin-auth";
import { normalizeCouponCode } from "@/lib/coupons";
import { db } from "@/lib/db";

export type CouponFormState = {
  error: string;
  success: string;
};

function field(formData: FormData, name: string): string {
  const value = formData.get(name);
  return typeof value === "string" ? value.trim() : "";
}

function amount(value: string): Prisma.Decimal | null {
  if (!/^\d{1,7}(?:\.\d{1,2})?$/.test(value)) {
    return null;
  }

  const result = new Prisma.Decimal(value);
  return result.lte(1000000) ? result : null;
}

// Date inputs are interpreted as Bangladesh calendar dates.
function bdDay(value: string, nextDay = false): Date | null {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return null;
  }

  const check = new Date(`${value}T00:00:00Z`);

  if (
    Number.isNaN(check.getTime()) ||
    check.toISOString().slice(0, 10) !== value
  ) {
    return null;
  }

  const date = new Date(`${value}T00:00:00+06:00`);

  if (nextDay) {
    date.setUTCDate(date.getUTCDate() + 1);
  }

  return date;
}

export async function createCoupon(
  _previous: CouponFormState,
  formData: FormData,
): Promise<CouponFormState> {
  const admin = await requireAdmin();

  if (admin.role !== "OWNER") {
    return {
      error: "Only the owner can manage coupons.",
      success: "",
    };
  }

  let code: string;

  try {
    code = normalizeCouponCode(field(formData, "code"));
  } catch {
    return {
      error: "Use 3–32 English letters, numbers, hyphens or underscores for the code.",
      success: "",
    };
  }

  const kind = field(formData, "kind");
  const value = amount(field(formData, "value"));
  const minimumSubtotal = amount(
    field(formData, "minimumSubtotal"),
  );
  const maximumText = field(formData, "maximumDiscount");
  const maximumDiscount = maximumText
    ? amount(maximumText)
    : null;
  const usageText = field(formData, "usageLimit");
  const usageLimit = usageText ? Number(usageText) : null;
  const startText = field(formData, "startsAt");
  const expiryText = field(formData, "expiresAt");
  const startsAt = startText ? bdDay(startText) : null;
  const expiresAt = expiryText
    ? bdDay(expiryText, true)
    : null;
  const isActive = formData.get("isActive") === "on";

  if (kind !== "PERCENT" && kind !== "FIXED") {
    return { error: "Choose a discount type.", success: "" };
  }

  if (
    !value ||
    value.lte(0) ||
    (kind === "PERCENT" && value.gt(100))
  ) {
    return {
      error:
        kind === "PERCENT"
          ? "Enter a percentage greater than 0 and up to 100."
          : "Enter a fixed discount greater than 0.",
      success: "",
    };
  }

  if (!minimumSubtotal || minimumSubtotal.lt(0)) {
    return {
      error: "Enter a valid minimum subtotal. Use 0 for no minimum.",
      success: "",
    };
  }

  if (
    maximumText &&
    (!maximumDiscount || maximumDiscount.lte(0))
  ) {
    return {
      error: "Enter a valid maximum discount or leave it empty.",
      success: "",
    };
  }

  if (
    usageText &&
    (!/^\d{1,7}$/.test(usageText) ||
      !Number.isSafeInteger(usageLimit) ||
      usageLimit! < 1 ||
      usageLimit! > 1000000)
  ) {
    return {
      error: "Usage limit must be a whole number from 1 to 1000000.",
      success: "",
    };
  }

  if ((startText && !startsAt) || (expiryText && !expiresAt)) {
    return {
      error: "Enter valid start and expiry dates.",
      success: "",
    };
  }

  if (
    startsAt &&
    expiresAt &&
    startsAt.getTime() >= expiresAt.getTime()
  ) {
    return {
      error: "Expiry date must be on or after the start date.",
      success: "",
    };
  }

  try {
    await db.coupon.create({
      data: {
        code,
        kind,
        value,
        minimumSubtotal,
        maximumDiscount,
        usageLimit,
        startsAt,
        expiresAt,
        isActive,
      },
    });
  } catch (error) {
    if (
      error instanceof Prisma.PrismaClientKnownRequestError &&
      error.code === "P2002"
    ) {
      return {
        error: "This coupon code already exists.",
        success: "",
      };
    }

    return {
      error: "Could not create the coupon. Please try again.",
      success: "",
    };
  }

  revalidatePath("/admin/coupons");

  return {
    error: "",
    success: `Coupon ${code} created.`,
  };
}

export async function toggleCoupon(
  formData: FormData,
): Promise<void> {
  const admin = await requireAdmin();

  if (admin.role !== "OWNER") return;

  const id = field(formData, "id");
  if (!id || id.length > 100) return;

  const coupon = await db.coupon.findUnique({
    where: { id },
    select: { isActive: true },
  });

  if (!coupon) return;

  await db.coupon.updateMany({
    where: { id, isActive: coupon.isActive },
    data: { isActive: !coupon.isActive },
  });

  revalidatePath("/admin/coupons");
}

class HeroInputError extends Error {}

function text(formData: FormData, key: string) {
  const value = formData.get(key);
  return typeof value === "string" ? value.trim() : "";
}

function validButtonHref(value: string): boolean {
  if (
    !value.startsWith("/") ||
    value.startsWith("//") ||
    /[\\\u0000-\u0020]/.test(value) ||
    value.length > 300
  ) {
    return false;
  }

  try {
    const url = new URL(value, "https://bimlay.com");

    if (url.origin !== "https://bimlay.com") {
      return false;
    }

    return (
      url.pathname === "/" ||
      url.pathname === "/shop" ||
      /^\/products\/[a-z0-9]+(?:-[a-z0-9]+)*$/.test(
        url.pathname,
      )
    );
  } catch {
    return false;
  }
}

async function verifyImage(url: string) {
  if (!url.startsWith("/media/")) {
    throw new HeroInputError("Use an uploaded banner image.");
  }

  const filename = url.slice("/media/".length);

  if (!MEDIA_FILENAME_PATTERN.test(filename)) {
    throw new HeroInputError("Invalid banner image.");
  }

  try {
  const exists = await mediaImageExists(url);

  if (!exists) {
    throw new Error("Image not found.");
  }
} catch {
  throw new HeroInputError(
    "A banner image is missing. Upload it again.",
  );
}
}

export async function saveHeroSlide(
  _previousState: HeroFormState,
  formData: FormData,
): Promise<HeroFormState> {
  const admin = await requireAdmin();

  if (admin.role !== "OWNER") {
    return {
      error: "Only the owner can manage banners.",
      success: "",
    };
  }

  const id = text(formData, "id");
  const eyebrow = text(formData, "eyebrow");
  const title = text(formData, "title");
  const subtitle = text(formData, "subtitle");
  const desktopImage = text(formData, "desktopImage");
  const mobileImage = text(formData, "mobileImage");
  const altText = text(formData, "altText");
  const buttonText = text(formData, "buttonText");
  const buttonHref = text(formData, "buttonHref");
  const orderText = text(formData, "sortOrder");
  const sortOrder = Number(orderText);
  const isActive = formData.get("isActive") === "on";

  let savedId = "";

  try {
    if (id.length > 100) {
      throw new HeroInputError("Invalid slide.");
    }

    if (
      title.length < 2 ||
      title.length > 90 ||
      eyebrow.length > 50 ||
      subtitle.length > 240
    ) {
      throw new HeroInputError(
        "Title: 2–90 characters. Eyebrow: maximum 50. Description: maximum 240.",
      );
    }

    if (altText.length < 2 || altText.length > 180) {
      throw new HeroInputError(
        "Describe the banner image in 2 to 180 characters.",
      );
    }

    if (buttonText.length < 2 || buttonText.length > 40) {
      throw new HeroInputError(
        "Button text must contain 2 to 40 characters.",
      );
    }

    if (!validButtonHref(buttonHref)) {
      throw new HeroInputError(
        "Use /shop, /shop?category=men, / or a valid /products/product-slug link.",
      );
    }

    if (
      orderText === "" ||
      !Number.isInteger(sortOrder) ||
      sortOrder < 0 ||
      sortOrder > 9999
    ) {
      throw new HeroInputError(
        "Display order must be a whole number from 0 to 9999.",
      );
    }

    await verifyImage(desktopImage);

    if (mobileImage) {
      await verifyImage(mobileImage);
    }

    savedId = await db.$transaction(
      async (tx) => {
        if (id) {
          const existing = await tx.heroSlide.findUnique({
            where: { id },
            select: { id: true },
          });

          if (!existing) {
            throw new HeroInputError("Slide not found.");
          }
        }

        if (isActive) {
          const activeCount = await tx.heroSlide.count({
            where: {
              isActive: true,
              ...(id ? { id: { not: id } } : {}),
            },
          });

          if (activeCount >= 8) {
            throw new HeroInputError(
              "Keep up to 8 active slides. Hide another slide first.",
            );
          }
        }

        const data = {
          eyebrow: eyebrow || null,
          title,
          subtitle: subtitle || null,
          desktopImage,
          mobileImage: mobileImage || null,
          altText,
          buttonText,
          buttonHref,
          sortOrder,
          isActive,
        };

        const slide = id
          ? await tx.heroSlide.update({
              where: { id },
              data,
              select: { id: true },
            })
          : await tx.heroSlide.create({
              data,
              select: { id: true },
            });

        return slide.id;
      },
      {
        isolationLevel:
          Prisma.TransactionIsolationLevel.Serializable,
      },
    );
  } catch (error) {
    if (error instanceof HeroInputError) {
      return { error: error.message, success: "" };
    }

    return {
      error: "Could not save the slide. Please try again.",
      success: "",
    };
  }

  revalidatePath("/");
  revalidatePath("/admin/hero");

  if (!id) {
    redirect(`/admin/hero?edit=${savedId}`);
  }

  return {
    error: "",
    success: "Slide updated successfully.",
  };
}

"use server";

import { revalidatePath } from "next/cache";
import { requireAdmin } from "@/lib/admin-auth";
import { db } from "@/lib/db";
import { Prisma } from "@/generated/prisma/client";
import {
  allowedPaymentStates,
  isOrderState,
  NEXT_ORDER_STATES,
} from "@/lib/order-rules";

export type OrderActionState = {
  error: string;
  success: string;
};

class OrderUpdateError extends Error {}

function text(formData: FormData, key: string) {
  const value = formData.get(key);
  return typeof value === "string" ? value.trim() : "";
}

function refreshOrderPages(orderId: string) {
  revalidatePath("/admin/orders");
  revalidatePath(`/admin/orders/${orderId}`);
  revalidatePath(`/order-confirmation/${orderId}`);
  revalidatePath("/admin/products");
  revalidatePath("/admin");
  revalidatePath("/shop");
  revalidatePath("/");
}

function errorState(error: unknown): OrderActionState {
  if (error instanceof OrderUpdateError) {
    return { error: error.message, success: "" };
  }

  if (
    error instanceof Prisma.PrismaClientKnownRequestError &&
    error.code === "P2034"
  ) {
    return {
      error: "Another update happened at the same time. Refresh and try again.",
      success: "",
    };
  }

  return {
    error: "Could not update this order. Please try again.",
    success: "",
  };
}

export async function changeOrderStatus(
  _previousState: OrderActionState,
  formData: FormData,
): Promise<OrderActionState> {
  const admin = await requireAdmin();

  if (admin.role !== "OWNER") {
    return {
      error: "Only the owner can change order status.",
      success: "",
    };
  }

  const orderId = text(formData, "orderId");
  const target = text(formData, "status");
  const reason = text(formData, "reason");
  const restockReturn = formData.get("restockReturn") === "on";

  if (!orderId || orderId.length > 100 || !isOrderState(target)) {
    return { error: "Invalid order or status.", success: "" };
  }

  if (reason.length > 500) {
    return {
      error: "Notes must not exceed 500 characters.",
      success: "",
    };
  }

  if (
    (target === "CANCELLED" || target === "RETURNED") &&
    reason.length < 5
  ) {
    return {
      error: "Enter a reason containing at least 5 characters.",
      success: "",
    };
  }

  try {
    await db.$transaction(
      async (tx) => {
        const order = await tx.order.findUnique({
          where: { id: orderId },
          include: { items: true },
        });

        if (!order) {
          throw new OrderUpdateError("Order not found.");
        }

        if (!NEXT_ORDER_STATES[order.status].includes(target)) {
          throw new OrderUpdateError(
            "This status change is not allowed. Refresh to see the current order status.",
          );
        }

        const shouldRestock =
          target === "CANCELLED" ||
          (target === "RETURNED" && restockReturn);

        let restoredUnits = 0;

        if (shouldRestock && !order.stockReleasedAt) {
          for (const item of order.items) {
            if (!item.variantId) continue;

            const result = await tx.productVariant.updateMany({
              where: { id: item.variantId },
              data: {
                stock: { increment: item.quantity },
              },
            });

            if (result.count === 1) {
              restoredUnits += item.quantity;
            }
          }
        }

        await tx.order.update({
          where: { id: orderId },
          data: {
            status: target,
            stockReleasedAt:
              shouldRestock && !order.stockReleasedAt
                ? new Date()
                : order.stockReleasedAt,
          },
        });

        const stockNote = shouldRestock
          ? `Restocked ${restoredUnits} units into existing variants.`
          : target === "RETURNED"
            ? "Returned items were not restocked."
            : "";

        await tx.orderAudit.create({
          data: {
            orderId,
            adminId: admin.id,
            actorEmail: admin.email,
            event: "STATUS_CHANGED",
            fromValue: order.status,
            toValue: target,
            note: [reason, stockNote].filter(Boolean).join(" "),
          },
        });
      },
      {
        isolationLevel:
          Prisma.TransactionIsolationLevel.Serializable,
        timeout: 15000,
      },
    );
  } catch (error) {
    return errorState(error);
  }

  refreshOrderPages(orderId);

  return {
    error: "",
    success: "Order status updated.",
  };
}

export async function changeOrderPayment(
  _previousState: OrderActionState,
  formData: FormData,
): Promise<OrderActionState> {
  const admin = await requireAdmin();

  if (admin.role !== "OWNER") {
    return {
      error: "Only the owner can record payments.",
      success: "",
    };
  }

  const orderId = text(formData, "orderId");
  const target = text(formData, "paymentStatus");
  const note = text(formData, "paymentNote");

  if (
    !orderId ||
    orderId.length > 100 ||
    (target !== "PAID" && target !== "REFUNDED")
  ) {
    return {
      error: "Invalid payment update.",
      success: "",
    };
  }

  if (note.length < 5 || note.length > 500) {
    return {
      error: "Enter a payment reference or note of 5 to 500 characters.",
      success: "",
    };
  }

  try {
    await db.$transaction(
      async (tx) => {
        const order = await tx.order.findUnique({
          where: { id: orderId },
        });

        if (!order) {
          throw new OrderUpdateError("Order not found.");
        }

        const allowed = allowedPaymentStates(
          order.status,
          order.paymentStatus,
        );

        if (!allowed.includes(target)) {
          throw new OrderUpdateError(
            "This payment change is not allowed for the current order.",
          );
        }

        await tx.order.update({
          where: { id: orderId },
          data: {
            paymentStatus: target,
          },
        });

        await tx.orderAudit.create({
          data: {
            orderId,
            adminId: admin.id,
            actorEmail: admin.email,
            event: "PAYMENT_CHANGED",
            fromValue: order.paymentStatus,
            toValue: target,
            note,
          },
        });
      },
      {
        isolationLevel:
          Prisma.TransactionIsolationLevel.Serializable,
      },
    );
  } catch (error) {
    return errorState(error);
  }

  refreshOrderPages(orderId);

  return {
    error: "",
    success: "Payment record updated.",
  };
}

function text(formData: FormData, key: string): string {
  const value = formData.get(key);

function objectValue(value: unknown): Record<string, unknown> {
  if (!value || typeof value !== "object" || Array.isArray(value)) {
    throw new ProductInputError("Invalid product data.");
  }

  return value as Record<string, unknown>;
}

function stringValue(value: unknown): string {
  return typeof value === "string" ? value.trim() : "";
}

function readArray(
  formData: FormData,
  key: string,
  maximum: number,
): unknown[] {
  const raw = text(formData, key);

  if (!raw || raw.length > 30000) {
    throw new ProductInputError("Invalid product data.");
  }

  let parsed: unknown;

  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new ProductInputError("Invalid product data.");
  }

  if (
    !Array.isArray(parsed) ||
    parsed.length === 0 ||
    parsed.length > maximum
  ) {
    throw new ProductInputError(
      key === "images"
        ? "Add between 1 and 8 product images."
        : "Add between 1 and 12 product variants.",
    );
  }

  return parsed;
}

function parsePrice(value: string): Prisma.Decimal {
  if (!/^\d{1,10}(?:\.\d{1,2})?$/.test(value)) {
    throw new ProductInputError(
      "Enter a valid price with up to 2 decimal places.",
    );
  }

  const price = new Prisma.Decimal(value);

  if (price.lte(0)) {
    throw new ProductInputError("Product price must be greater than zero.");
  }

  return price;
}

type SubmittedImage = {

function parseSubmittedImages(
  formData: FormData,
  status: string,
): SubmittedImage[] {
  if (formData.get("imagesUploading") === "1") {
    throw new ProductEditError(
      "Wait for the image upload to finish, then save again.",
    );
  }

  const raw = formData.get("images");

  if (typeof raw !== "string" || raw.length > 16000) {
    throw new ProductEditError("Invalid image list.");
  }

  let decoded: unknown;

  try {
    decoded = JSON.parse(raw);
  } catch {
    throw new ProductEditError("Invalid image list.");
  }

  if (!Array.isArray(decoded) || decoded.length > 8) {
    throw new ProductEditError("A product can have up to 8 images.");
  }

  if (status === "ACTIVE" && decoded.length === 0) {
    throw new ProductEditError("An active product needs at least one image.");
  }

  const seenIds = new Set<string>();
  const seenUrls = new Set<string>();

  return decoded.map((item): SubmittedImage => {
    if (!item || typeof item !== "object" || typeof item.alt !== "string") {
      throw new ProductEditError("Invalid image information.");
    }

    const alt = item.alt.trim();

    if (alt.length < 2 || alt.length > 180) {
      throw new ProductEditError(
        "Each image description must contain 2 to 180 characters.",
      );
    }

    if ("id" in item) {
      if (
        typeof item.id !== "string" ||
        !item.id ||
        item.id.length > 100 ||
        seenIds.has(item.id) ||
        ("url" in item && typeof item.url !== "string")
      ) {
        throw new ProductEditError("Invalid existing image.");
      }

      seenIds.add(item.id);
      return { id: item.id, alt };
    }

    if (
      !("url" in item) ||
      typeof item.url !== "string" ||
      seenUrls.has(item.url)
    ) {
      throw new ProductEditError("Invalid uploaded image.");
    }

    const match =
      /^\/media\/([0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\.webp)$/i.exec(
        item.url,
      );

    if (!match) {
      throw new ProductEditError("Invalid uploaded image URL.");
    }

    seenUrls.add(item.url);

    return {
      url: item.url,
      alt,
    };
  });
}

                sku: true,
                size: true,
                color: true,
              },
            },
            images: {
              orderBy: [{ sortOrder: "asc" }, { id: "asc" }],
              select: {
                id: true,
              },
            },
          },
        });
        if (!product) {
          throw new ProductEditError("Product was not found.");
        }

        if (product.variants.length + newVariants.length > 12) {
          throw new ProductEditError("A product can have up to 12 variants.");
        }

        const allSkus = [
          ...product.variants.map((variant) => variant.sku),
          ...newVariants.map((variant) => variant.sku),
        ];

        if (new Set(allSkus).size !== allSkus.length) {
          throw new ProductEditError("Each variant must have a different SKU.");
        }

        const combinations = [
          ...product.variants.map((variant) =>
            JSON.stringify([
              variant.size?.trim().toLowerCase() ?? "",
              variant.color?.trim().toLowerCase() ?? "",
            ]),
          ),
          ...newVariants.map((variant) =>
            JSON.stringify([
              variant.size?.toLowerCase() ?? "",
              variant.color?.toLowerCase() ?? "",
            ]),
          ),
        ];

        if (new Set(combinations).size !== combinations.length) {
          throw new ProductEditError(
            "Each size and colour combination must be different.",
          );
        }

        const databaseImageIds = product.images.map((image) => image.id);

        if (
          JSON.stringify(databaseImageIds) !== JSON.stringify(originalImageIds)
        ) {
          throw new ProductEditError(
            "Product images changed in another window. Refresh and try again.",
          );
        }

        const ownedImageIds = new Set(databaseImageIds);

        for (const image of submittedImages) {
          if (image.id && !ownedImageIds.has(image.id)) {
            throw new ProductEditError(
              "An image does not belong to this product.",
            );
          }
        }

        const category = await tx.category.findUnique({
          where: { id: categoryId },
          select: {
            isActive: true,
            parent: {
              select: { isActive: true },
            },
          },
        });

        if (!category) {
          throw new ProductEditError("Category was not found.");
        }

        if (
          status === "ACTIVE" &&
          (!category.isActive || (category.parent && !category.parent.isActive))
        ) {
          throw new ProductEditError(
            "An active product must belong to an active category with an active parent.",
          );
        }

        const updates = product.variants.map((variant) => {
          const price = priceValue(text(formData, `price-${variant.id}`));

          const regularText = text(formData, `regular-${variant.id}`);

          const compareAtPrice = regularText ? priceValue(regularText) : null;

          const stockText = text(formData, `stock-${variant.id}`);

          if (!/^\d{1,7}$/.test(stockText) || Number(stockText) > 1000000) {
            throw new ProductEditError(
              `${variant.sku}: stock must be a whole number from 0 to 1000000.`,
            );
          }

          if (compareAtPrice && compareAtPrice.lte(price)) {
            throw new ProductEditError(
              `${variant.sku}: regular price must exceed selling price, or be empty.`,
            );
          }

          return {
            id: variant.id,
            price,
            compareAtPrice,
            stock: Number(stockText),
            isActive: formData.get(`active-${variant.id}`) === "on",
          };
        });

        if (
          status === "ACTIVE" &&
          !updates.some((variant) => variant.isActive) &&
          !newVariants.some((variant) => variant.isActive)
        ) {
          throw new ProductEditError(
            "An active product needs at least one active variant.",
          );
        }

        await tx.product.update({
          where: { id: productId },
          data: {
            name,
            categoryId,
            shortDescription: shortDescription || null,
            description,
            status,
            isFeatured,
          },
        });

        for (const variant of updates) {
          await tx.productVariant.update({
            where: { id: variant.id },
            data: {
              price: variant.price,
              compareAtPrice: variant.compareAtPrice,
              stock: variant.stock,
              isActive: variant.isActive,
            },
          });
        }

        for (const variant of newVariants) {
          await tx.productVariant.create({
            data: {
              productId,
              sku: variant.sku,
              size: variant.size,
              color: variant.color,
              price: variant.price,
              compareAtPrice: variant.compareAtPrice,
              stock: variant.stock,
              isActive: variant.isActive,
            },
          });
        }

        const retainedIds = submittedImages
          .filter((image) => image.id)
          .map((image) => image.id as string);

        await tx.productImage.deleteMany({
          where: {
            productId,
            id: {
              notIn: retainedIds,
            },
          },
        });

        for (const [index, image] of submittedImages.entries()) {
          if (image.id) {
            await tx.productImage.update({
              where: { id: image.id },
              data: {
                alt: image.alt,
                sortOrder: index,
              },
            });
          } else if (image.url) {
            await tx.productImage.create({
              data: {
                productId,
                url: image.url,
                alt: image.alt,
                sortOrder: index,
              },
            });
          }
        }
      },
      {
        isolationLevel: Prisma.TransactionIsolationLevel.Serializable,
      },
    );
  } catch (error) {
    if (error instanceof ProductEditError) {
      return { error: error.message, success: "" };
    }

    if (
  error instanceof Prisma.PrismaClientKnownRequestError &&
  error.code === "P2002"
) {
  return {
    error: "This SKU is already used by another product.",
    success: "",
  };
}

    if (
      error instanceof Prisma.PrismaClientKnownRequestError &&
      error.code === "P2034"
    ) {
      return {
        error: "Another update happened at the same time. Please save again.",
        success: "",
      };
    }

    return {
      error: "Could not update the product. Please try again.",
      success: "",
    };
  }

  revalidatePath("/admin/products");
  revalidatePath(`/admin/products/${productId}/edit`);
  revalidatePath("/admin");
  revalidatePath("/");
  revalidatePath("/shop");

  return {
    error: "",
    success: "Product updated successfully.",
  };
}

export default function ProductImageEditor({
  images: initialImages,
  productName,
}: {
  images: ExistingImage[];
  productName: string;
}) {
  const [images, setImages] = useState<EditableImage[]>(
    initialImages.map((image) => ({
      id: image.id,
      url: image.url,
      alt: image.alt || productName,
    })),
  );

  const [uploading, setUploading] = useState(false);
  const [error, setError] = useState("");
  const fileRef = useRef<HTMLInputElement>(null);

  async function uploadImage(
    event: React.ChangeEvent<HTMLInputElement>,
  ) {
    const file = event.target.files?.[0];

    if (!file) return;

    if (images.length >= 8) {
      setError("A product can have up to 8 images.");
      event.target.value = "";
      return;
    }

    if (
      !["image/jpeg", "image/png", "image/webp"].includes(file.type) ||
      file.size > 8 * 1024 * 1024
    ) {
      setError("Choose a JPG, PNG or WebP image up to 8 MB.");
      event.target.value = "";
      return;
    }

    setError("");
    setUploading(true);

    try {
      const body = new FormData();
      body.append("image", file);

      const response = await fetch("/api/admin/uploads", {
        method: "POST",
        body,
      });

      const result: unknown = await response.json();

      if (!result || typeof result !== "object") {
        throw new Error("Unexpected upload response.");
      }

      if (!response.ok) {
        const message =
          "error" in result && typeof result.error === "string"
            ? result.error
            : "Image upload failed.";

        throw new Error(message);
      }

      if (
        !("url" in result) ||
        typeof result.url !== "string" ||
        !/^\/media\/[0-9a-f-]+\.webp$/i.test(result.url)
      ) {
        throw new Error("The upload did not return a valid image.");
      }

      setImages((current) => [
        ...current,
        {
          url: result.url as string,
          alt: productName,
        },
      ]);
    } catch (cause) {
      setError(
        cause instanceof Error
          ? cause.message
          : "Image upload failed.",
      );
    } finally {
      setUploading(false);

      if (fileRef.current) {
        fileRef.current.value = "";
      }
    }
  }

  function moveImage(index: number, direction: -1 | 1) {
    setImages((current) => {
      const destination = index + direction;

      if (destination < 0 || destination >= current.length) {
        return current;
      }

      const next = [...current];

      [next[index], next[destination]] = [
        next[destination],
        next[index],
      ];

      return next;
    });
  }

  return (
    <section className="category-panel">
      <h2 className="product-section-title">Product images</h2>

      <p className="category-help">
        The first image is the main product image. Add up to 8
        images, then save the product to keep your changes.
      </p>

      <input
        type="hidden"
        name="images"
        value={JSON.stringify(images)}
      />

      <input
        type="hidden"
        name="imagesOriginal"
        value={JSON.stringify(
          initialImages.map((image) => image.id),
        )}
      />

      <input
        type="hidden"
        name="imagesUploading"
        value={uploading ? "1" : "0"}
      />

      <div className="product-edit-images">
        {images.map((image, index) => (
          <div
            className="product-edit-image"
            key={image.id ?? image.url}
          >
            <div className="product-edit-image-preview">
              <Image
                src={image.url}
                alt={image.alt}
                width={220}
                height={220}
                unoptimized
              />
            </div>

            <p className="category-help">
              {index === 0
                ? "Main image"
                : `Image ${index + 1}`}
            </p>

            <label>
              <span>Image description (alt text)</span>
              <input
                value={image.alt}
                onChange={(event) => {
                  const alt = event.target.value;

                  setImages((current) =>
                    current.map((item, itemIndex) =>
                      itemIndex === index
                        ? { ...item, alt }
                        : item,
                    ),
                  );
                }}
                minLength={2}
                maxLength={180}
                required
              />
            </label>

            <div className="product-edit-image-buttons">
              <button
                type="button"
                onClick={() => moveImage(index, -1)}
                disabled={index === 0 || uploading}
              >
                Move up
              </button>

              <button
                type="button"
                onClick={() => moveImage(index, 1)}
                disabled={
                  index === images.length - 1 || uploading
                }
              >
                Move down
              </button>

              <button
                type="button"
                onClick={() =>
                  setImages((current) =>
                    current.filter(
                      (_, itemIndex) => itemIndex !== index,
                    ),
                  )
                }
                disabled={uploading}
              >
                Remove
              </button>
            </div>
          </div>
        ))}
      </div>

      <label className="product-edit-upload">
        <span>Add product image</span>
        <input
          ref={fileRef}
          type="file"
          accept="image/jpeg,image/png,image/webp"
          onChange={uploadImage}

      <p className="category-help">

"use server";

import { revalidatePath } from "next/cache";
import { requireAdmin } from "@/lib/admin-auth";
import { db } from "@/lib/db";
import { Prisma } from "@/generated/prisma/client";

export type SettingsState = {
  error: string;
  success: string;
};

function text(formData: FormData, key: string): string {
  const value = formData.get(key);
  return typeof value === "string" ? value.trim() : "";
}

function fail(error: string): SettingsState {
  return { error, success: "" };
}

function parseCharge(value: string): Prisma.Decimal | null {
  if (!/^\d{1,7}(?:\.\d{1,2})?$/.test(value)) {
    return null;
  }

  const amount = new Prisma.Decimal(value);

  if (amount.lt(0) || amount.gt(1000000)) {
    return null;
  }

  return amount;
}

function normalizeWhatsApp(value: string): string {
  const digits = value
    .replace(/[০-৯]/g, (digit) =>
      String("০১২৩৪৫৬৭৮৯".indexOf(digit)),
    )
    .replace(/[\s()+-]/g, "");

  return digits.startsWith("01") ? `88${digits}` : digits;
}

function normalizeFacebook(value: string): string | null {
  if (!value) return "";

  if (value.length > 500) return null;

  try {
    const url = new URL(value);

    const allowedHosts = new Set([
      "facebook.com",
      "www.facebook.com",
      "m.facebook.com",
      "fb.com",
      "www.fb.com",
    ]);

    if (
      url.protocol !== "https:" ||
      !allowedHosts.has(url.hostname.toLowerCase()) ||
      url.username ||
      url.password ||
      url.port
    ) {
      return null;
    }

    url.hash = "";

    return url.toString();
  } catch {
    return null;
  }
}

export async function saveStoreSettings(
  _previousState: SettingsState,
  formData: FormData,
): Promise<SettingsState> {
  const admin = await requireAdmin();

  if (admin.role !== "OWNER") {
    return fail("Only the owner can change store settings.");
  }

  const storeName = text(formData, "storeName");

  const supportPhone = text(formData, "supportPhone")
    .replace(/[\s()-]/g, "");

  const whatsappNumber = normalizeWhatsApp(
    text(formData, "whatsappNumber"),
  );

  const facebookPageUrl = normalizeFacebook(
    text(formData, "facebookPageUrl"),
  );

  const supportEmail = text(formData, "supportEmail");

  const dhakaDeliveryCharge = parseCharge(
    text(formData, "dhakaDeliveryCharge"),
  );

  const outsideDeliveryCharge = parseCharge(
    text(formData, "outsideDeliveryCharge"),
  );

  const codEnabled = formData.get("codEnabled") === "on";

  if (storeName.length < 2 || storeName.length > 100) {
    return fail("Store name must contain 2 to 100 characters.");
  }

  if (supportPhone && !/^\+?\d{7,15}$/.test(supportPhone)) {
    return fail("Enter a valid support phone number or leave it empty.");
  }

  if (
    whatsappNumber &&
    !/^8801[3-9]\d{8}$/.test(whatsappNumber)
  ) {
    return fail(
      "Enter a valid Bangladesh WhatsApp number starting with 01 or 8801.",
    );
  }

  if (facebookPageUrl === null) {
    return fail(
      "Enter a complete HTTPS Facebook page link, or leave it empty.",
    );
  }

  if (
    supportEmail &&
    (
      supportEmail.length > 254 ||
      !/^[^\s@<>]+@[^\s@<>]+\.[^\s@<>]+$/.test(supportEmail)
    )
  ) {
    return fail("Enter a valid support email or leave it empty.");
  }

  if (!dhakaDeliveryCharge || !outsideDeliveryCharge) {
    return fail(
      "Enter valid delivery charges from 0 to 1000000, with up to 2 decimal places.",
    );
  }

  const data = {
    storeName,
    supportPhone: supportPhone || null,
    whatsappNumber: whatsappNumber || null,
    facebookPageUrl: facebookPageUrl || null,
    supportEmail: supportEmail || null,
    dhakaDeliveryCharge,
    outsideDeliveryCharge,
    codEnabled,
  };

  try {
    await db.storeSetting.upsert({
      where: { id: "main" },
      create: {
        id: "main",
        ...data,
      },
      update: data,
    });
  } catch {
    return fail("Could not save settings. Please try again.");
  }

  revalidatePath("/admin/settings");
  revalidatePath("/checkout");
  revalidatePath("/", "layout");

  return {
    error: "",
    success: "Store settings saved successfully.",
  };
}

import { randomUUID } from "node:crypto";
import sharp, { type OutputInfo } from "sharp";
import { getAdmin } from "@/lib/admin-auth";
import { saveMediaImage } from "@/lib/media-storage";

const MAX_FILE_BYTES = 8 * 1024 * 1024;
const MAX_REQUEST_BYTES = 10 * 1024 * 1024;
const MAX_PIXELS = 40_000_000;

const ALLOWED_MIME_TYPES = new Set([
  "image/jpeg",
  "image/png",
  "image/webp",
]);

class UploadTooLargeError extends Error {}

function isAllowedOrigin(request: Request): boolean {
  const origin = request.headers.get("origin");
  if (!origin) return true; // Direct/same-origin server-side request

  // Render Environment Variable চেক
  if (process.env.APP_ORIGIN && origin === process.env.APP_ORIGIN) {
    return true;
  }

  try {
    const originUrl = new URL(origin);
    const hostHeader =
      request.headers.get("x-forwarded-host") ??
      request.headers.get("host") ??
      new URL(request.url).host;

    // একই হোস্টের রিকোয়েস্ট হলে এলাউ করবে
    if (originUrl.host === hostHeader) {
      return true;
    }

    // bimlay.com এবং এর সাবডোমেইন এলাউ করতে
    if (
      originUrl.hostname === "bimlay.com" ||
      originUrl.hostname.endsWith(".bimlay.com") ||
      originUrl.hostname === "localhost" ||
      originUrl.hostname === "127.0.0.1" ||
      originUrl.hostname.endsWith(".onrender.com")
    ) {
      return true;
    }
  } catch {
    return false;
  }

  return false;
}

async function readLimitedFormData(request: Request): Promise<FormData> {
  const contentType = request.headers.get("content-type") ?? "";

  if (!contentType.toLowerCase().startsWith("multipart/form-data;")) {
    throw new Error("Invalid upload request.");
  }

  const declaredSize = Number(request.headers.get("content-length") ?? 0);

  if (declaredSize > MAX_REQUEST_BYTES) {
    throw new UploadTooLargeError();
  }

  if (!request.body) {
    throw new Error("Missing upload body.");
  }

  const reader = request.body.getReader();
  const chunks: Uint8Array[] = [];
  let total = 0;

  try {
    while (true) {
      const { done, value } = await reader.read();

      if (done) {
        break;
      }

      total += value.byteLength;

      if (total > MAX_REQUEST_BYTES) {
        await reader.cancel();
        throw new UploadTooLargeError();
      }

      chunks.push(value);
    }
  } finally {
    reader.releaseLock();
  }

  const body = new Uint8Array(total);
  let offset = 0;

  for (const chunk of chunks) {
    body.set(chunk, offset);
    offset += chunk.byteLength;
  }

  return new Response(body.buffer, {
    headers: {
      "Content-Type": contentType,
    },
  }).formData();
}

export async function POST(request: Request) {
  if (!isAllowedOrigin(request)) {
    return Response.json(
      { error: "Upload origin is not allowed." },
      { status: 403 },
    );
  }

  const admin = await getAdmin();

  if (!admin || admin.role !== "OWNER") {
    return Response.json(
      { error: "Owner login is required." },
      { status: 403 },
    );
  }

  let formData: FormData;

  try {
    formData = await readLimitedFormData(request);
  } catch (error) {
    if (error instanceof UploadTooLargeError) {
      return Response.json(
        { error: "Upload is too large. Choose an image under 8 MB." },
        { status: 413 },
      );
    }

    return Response.json(
      { error: "Invalid upload request." },
      { status: 400 },
    );
  }

  const images = formData.getAll("image");
  const file = images[0];

  if (images.length !== 1 || !(file instanceof File)) {
    return Response.json(
      { error: "Choose one image." },
      { status: 400 },
    );
  }

  if (file.size === 0 || file.size > MAX_FILE_BYTES) {
    return Response.json(
      { error: "Image must be smaller than 8 MB and not empty." },
      { status: 400 },
    );
  }

  if (!ALLOWED_MIME_TYPES.has(file.type)) {
    return Response.json(
      { error: "Only JPG, PNG and WebP images are supported." },
      { status: 400 },
    );
  }

  let processed: {
    data: Buffer;
    info: OutputInfo;
  };

  try {
    const input = Buffer.from(await file.arrayBuffer());

    const metadata = await sharp(input, {
      limitInputPixels: MAX_PIXELS,
    }).metadata();

    if (
      !metadata.format ||
      !["jpeg", "png", "webp"].includes(metadata.format) ||
      !metadata.width ||
      !metadata.height ||
      (metadata.pages ?? 1) > 1
    ) {
      return Response.json(
        { error: "Choose a valid, non-animated JPG, PNG or WebP image." },
        { status: 400 },
      );
    }

    processed = await sharp(input, {
      limitInputPixels: MAX_PIXELS,
    })
      .rotate()
      .resize({
        width: 1600,
        height: 1600,
        fit: "inside",
        withoutEnlargement: true,
      })
      .webp({ quality: 85 })
      .toBuffer({ resolveWithObject: true });
  } catch {
    return Response.json(
      {
        error:
          "This image could not be processed. Choose a valid image under 40 megapixels.",
      },
      { status: 400 },
    );
  }

  const filename = `${randomUUID()}.webp`;

  try {
    await saveMediaImage(filename, processed.data);
  } catch {
    return Response.json(
      { error: "Could not save the image. Please try again." },
      { status: 500 },
    );
  }

  return Response.json(
    {
      url: `/media/${filename}`,
      width: processed.info.width,
      height: processed.info.height,
      bytes: processed.info.size,
    },
    {
      status: 201,
      headers: {
        "Cache-Control": "no-store",
      },
    },
  );
}

function single(value: string | string[] | undefined) {

function priceValue(value: string): number | undefined {
  if (!/^\d{1,7}(?:\.\d{1,2})?$/.test(value)) {
    return undefined;
  }

  const price = Number(value);
  return price <= 1000000 ? price : undefined;
}

async function ShopContent({
  searchParams,

}) {

  const page =
    Number.isSafeInteger(pageNumber) && pageNumber > 0 ? pageNumber : 1;

  const minPrice = priceValue(single(params.minPrice));
  const maxPrice = priceValue(single(params.maxPrice));
  const size = single(params.size).trim().slice(0, 40);

    page,
    minPrice,

        : maxPrice,
    size,
    color,
    inStockOnly,
    sort,

import { useEffect, useState } from "node_modules/@types/react"; // অথবা শুধু 'react'

export default function FlashTimer({ endTime }: { endTime: string }) {
  const [timeLeft, setTimeLeft] = useState({
    hours: 0,
    minutes: 0,
    seconds: 0,
    isExpired: false,
  });

  useEffect(() => {
    const targetTime = new Date(endTime).getTime();

    const interval = setInterval(() => {
      const now = new Date().getTime();
      const difference = targetTime - now;

      if (difference <= 0) {
        clearInterval(interval);
        setTimeLeft({ hours: 0, minutes: 0, seconds: 0, isExpired: true });
      } else {
        const hours = Math.floor((difference / (1000 * 60 * 60)));
        const minutes = Math.floor((difference % (1000 * 60 * 60)) / (1000 * 60));
        const seconds = Math.floor((difference % (1000 * 60)) / 1000);

        setTimeLeft({ hours, minutes, seconds, isExpired: false });
      }
    }, 1000);

    return () => clearInterval(interval);
  }, [endTime]);

  if (timeLeft.isExpired) {
    return <span style={{ color: "red", fontWeight: "bold" }}>Flash Sale Ended</span>;
  }

  return (
    <div style={{ display: "flex", gap: "0.5rem", alignItems: "center", background: "#fee2e2", padding: "0.5rem 1rem", borderRadius: "6px", color: "#991b1b", width: "fit-content" }}>
      <span style={{ fontSize: "0.85rem", fontWeight: "600" }}>Limited Time Deal:</span>
      <div style={{ fontFamily: "monospace", fontWeight: "bold" }}>
        {String(timeLeft.hours).padStart(2, "0")}h : {String(timeLeft.minutes).padStart(2, "0")}m : {String(timeLeft.seconds).padStart(2, "0")}s
      </div>
    </div>
  );
}

"use client";

import Link from "next/link";
import {
  useEffect,
  useRef,
  useState,
  type PointerEvent,
  type ReactNode,
} from "react";
import { ShoppingBag } from "lucide-react";

type Position = {
  x: number;
  y: number;
};

type DragState = {
  pointerId: number;
  startX: number;
  startY: number;
  left: number;
  top: number;
};

function keepInsideScreen(
  position: Position,
  width: number,
  height: number,
): Position {
  const padding = 12;
  const bottomSpace = window.matchMedia("(max-width: 767px)").matches
    ? 90
    : 16;

  return {
    x: Math.max(
      padding,
      Math.min(position.x, window.innerWidth - width - padding),
    ),
    y: Math.max(
      padding,
      Math.min(position.y, window.innerHeight - height - bottomSpace),
    ),
  };
}

function DraggableAction({
  href,
  label,
  variant,
  external = false,
  children,
}: {
  href: string;
  label: string;
  variant: "cart" | "whatsapp";
  external?: boolean;
  children: ReactNode;
}) {
  const elementRef = useRef<HTMLAnchorElement>(null);
  const dragRef = useRef<DragState | null>(null);
  const blockClickRef = useRef(false);

  const [position, setPosition] = useState<Position | null>(null);
  const [dragging, setDragging] = useState(false);

  useEffect(() => {
    function handleResize() {
      const element = elementRef.current;
      if (!element) return;

      const { width, height } = element.getBoundingClientRect();

      setPosition((current) =>
        current ? keepInsideScreen(current, width, height) : current,
      );
    }

    window.addEventListener("resize", handleResize);

    return () => window.removeEventListener("resize", handleResize);
  }, []);

  function startDrag(event: PointerEvent<HTMLAnchorElement>) {
    if (!event.isPrimary || event.button !== 0) return;

    const rect = event.currentTarget.getBoundingClientRect();

    blockClickRef.current = false;

    dragRef.current = {
      pointerId: event.pointerId,
      startX: event.clientX,
      startY: event.clientY,
      left: rect.left,
      top: rect.top,
    };

    event.currentTarget.setPointerCapture(event.pointerId);
  }

  function moveDrag(event: PointerEvent<HTMLAnchorElement>) {
    const drag = dragRef.current;

    if (!drag || drag.pointerId !== event.pointerId) return;

    const dx = event.clientX - drag.startX;
    const dy = event.clientY - drag.startY;

    // Small finger movements still count as a normal tap.
    if (!blockClickRef.current && Math.hypot(dx, dy) < 8) return;

    blockClickRef.current = true;
    setDragging(true);

    const { width, height } = event.currentTarget.getBoundingClientRect();

    setPosition(
      keepInsideScreen(
        {
          x: drag.left + dx,
          y: drag.top + dy,
        },
        width,
        height,
      ),
    );
  }

  function finishDrag(event: PointerEvent<HTMLAnchorElement>) {
    if (dragRef.current?.pointerId !== event.pointerId) return;

    dragRef.current = null;
    setDragging(false);

    if (event.currentTarget.hasPointerCapture(event.pointerId)) {
      event.currentTarget.releasePointerCapture(event.pointerId);
    }
  }

  return (
    <Link
      ref={elementRef}
      href={href}
      prefetch={false}
      target={external ? "_blank" : undefined}
      rel={external ? "noopener noreferrer" : undefined}
      aria-label={label}
      title={`${label} — ধরে সরাতে পারবে`}
      draggable={false}
      className={[
        "bimlay-floating-action",
        `bimlay-floating-${variant}`,
        dragging ? "is-dragging" : "",
      ].join(" ")}
      style={
        position
          ? {
              left: position.x,
              top: position.y,
              right: "auto",
              bottom: "auto",
            }
          : undefined
      }
      onPointerDown={startDrag}
      onPointerMove={moveDrag}
      onPointerUp={finishDrag}
      onPointerCancel={finishDrag}
      onLostPointerCapture={() => {
        dragRef.current = null;
        setDragging(false);
      }}
      onDragStart={(event) => event.preventDefault()}
      onClick={(event) => {
        if (blockClickRef.current) {
          event.preventDefault();
          event.stopPropagation();
        }
      }}
      onKeyDown={(event) => {
        if (event.key === "Enter") {
          blockClickRef.current = false;
          return;
        }

        // Home restores the original position.
        if (event.key === "Home") {
          event.preventDefault();
          setPosition(null);
          return;
        }

        const directions: Record<string, Position> = {
          ArrowLeft: { x: -16, y: 0 },
          ArrowRight: { x: 16, y: 0 },
          ArrowUp: { x: 0, y: -16 },
          ArrowDown: { x: 0, y: 16 },
        };

        const direction = directions[event.key];
        if (!direction) return;

        event.preventDefault();

        const rect = event.currentTarget.getBoundingClientRect();

        setPosition(
          keepInsideScreen(
            {
              x: rect.left + direction.x,
              y: rect.top + direction.y,
            },
            rect.width,
            rect.height,
          ),
        );
      }}
    >
      {children}
    </Link>
  );
}

"use client";

import { useRef, useState } from "react";

function productUrl() {
  const canonical = document.querySelector<HTMLLinkElement>(
    'link[rel="canonical"]',
  )?.href;

  return canonical || new URL(
    window.location.pathname,
    window.location.origin,
  ).href;
}

export default function ProductShare() {
  const [message, setMessage] = useState("");
  const [manualUrl, setManualUrl] = useState("");
  const manualInput = useRef<HTMLInputElement>(null);

  function openShare(url: string) {
    window.open(url, "_blank", "noopener,noreferrer");
  }

  async function copyLink() {
    const url = productUrl();

    try {
      if (!navigator.clipboard?.writeText) {
        throw new Error("Clipboard unavailable");
      }

      await navigator.clipboard.writeText(url);
      setManualUrl("");
      setMessage("Product link copied.");
    } catch {
      setManualUrl(url);
      setMessage("Select and copy the link below.");

      window.requestAnimationFrame(() => {
        manualInput.current?.focus();
        manualInput.current?.select();
      });
    }
  }

  async function nativeShare() {
    if (!navigator.share) {
      await copyLink();
      return;
    }

    try {
      await navigator.share({
        title: document.title,
        url: productUrl(),
      });
      setMessage("");
    } catch (error) {
      if (
        error instanceof DOMException &&
        error.name === "AbortError"
      ) {
        return;
      }

      await copyLink();
    }
  }

  return (
    <section className="product-share" aria-label="Share this product">
      <span className="product-share-label">Share this product</span>

      <div className="product-share-actions">
        <button
          type="button"
          onClick={() => {
            const text = `Take a look at this product: ${productUrl()}`;
            openShare(`https://wa.me/?text=${encodeURIComponent(text)}`);
          }}
        >
          WhatsApp
        </button>

        <button
          type="button"
          onClick={() => {
            openShare(
              `https://www.facebook.com/sharer/sharer.php?u=${
                encodeURIComponent(productUrl())
              }`,
            );
          }}
        >
          Facebook
        </button>

        <button type="button" onClick={copyLink}>
          Copy link
        </button>

        <button type="button" onClick={nativeShare}>
          Share
        </button>
      </div>

      <p className="product-share-message" role="status">
        {message}
      </p>

      {manualUrl && (
        <input
          ref={manualInput}
          type="text"
          readOnly
          value={manualUrl}
          aria-label="Product link to copy"
          onFocus={(event) => event.currentTarget.select()}
        />
      )}
    </section>
  );
}

"use client";

import Image from "next/image";
import { useRef, useState } from "react";
import { Upload } from "lucide-react";

type UploadedImage = {
  url: string;
  width: number;
  height: number;
  bytes: number;
};

type Props = {
  onUploaded?: (image: UploadedImage) => void;
};

export default function ImageUpload({ onUploaded }: Props) {
  const inputRef = useRef<HTMLInputElement>(null);

  const [pending, setPending] = useState(false);
  const [error, setError] = useState("");
  const [image, setImage] = useState<UploadedImage | null>(null);

  async function uploadImage() {
    const file = inputRef.current?.files?.[0];

    setError("");

    if (!file) {
      setError("Choose an image first.");
      return;
    }

    if (
      !["image/jpeg", "image/png", "image/webp"].includes(file.type)
    ) {
      setError("Choose a JPG, PNG or WebP image.");
      return;
    }

    if (file.size === 0 || file.size > 8 * 1024 * 1024) {
      setError("Choose a non-empty image smaller than 8 MB.");
      return;
    }

    setPending(true);

    try {
      const formData = new FormData();
      formData.set("image", file);

      const response = await fetch("/api/admin/uploads", {
        method: "POST",
        body: formData,
      });

      const result = await response.json();

      if (!response.ok) {
        throw new Error(
          typeof result.error === "string"
            ? result.error
            : "Upload failed.",
        );
      }

      if (
        typeof result.url !== "string" ||
        typeof result.width !== "number" ||
        typeof result.height !== "number" ||
        typeof result.bytes !== "number"
      ) {
        throw new Error("Unexpected upload response.");
      }

      const uploaded: UploadedImage = {
        url: result.url,
        width: result.width,
        height: result.height,
        bytes: result.bytes,
      };

      setImage(uploaded);
      onUploaded?.(uploaded);

      if (inputRef.current) {
        inputRef.current.value = "";
      }
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : "Upload failed. Please try again.",
      );
    } finally {
      setPending(false);
    }
  }

  return (
    <div className="image-upload">
      <label className="image-upload-label">
        <span>Choose an image</span>

        <input
          ref={inputRef}
          type="file"
          accept="image/jpeg,image/png,image/webp"
          disabled={pending}
        />
      </label>

      <p className="category-help">
        JPG, PNG or WebP · Maximum 8 MB · Automatically optimised
      </p>

      <button
        type="button"
        className="primary-button"
        onClick={uploadImage}
        disabled={pending}
      >
        <Upload size={16} />
        {pending ? "Uploading..." : "Upload image"}
      </button>

      <div aria-live="polite" aria-atomic="true">
        {error && <p className="admin-form-error">{error}</p>}

        {image && (
          <div className="image-upload-result">
            <p className="category-success">
              Image uploaded successfully.
            </p>

            <div className="image-upload-preview">
              <Image
                src={image.url}
                alt="Uploaded image preview"
                width={image.width}
                height={image.height}
                unoptimized
              />
            </div>

            <p className="category-help">
              {image.width} × {image.height} px ·{" "}
              {Math.ceil(image.bytes / 1024)} KB · WebP

import { createHash, randomBytes } from "node:crypto";
import { cookies } from "next/headers";
import { redirect } from "next/navigation";
import { db } from "@/lib/db";

const COOKIE_NAME = "bimlay_admin_session";
const SESSION_SECONDS = 8 * 60 * 60;

function tokenHash(token: string) {
  return createHash("sha256").update(token).digest("hex");
}

export async function createAdminSession(adminId: string) {
  const cookieStore = await cookies();

  const previousToken = cookieStore.get(COOKIE_NAME)?.value;

  if (previousToken && /^[a-f0-9]{64}$/.test(previousToken)) {
    await db.adminSession.deleteMany({
      where: {
        tokenHash: tokenHash(previousToken),
      },
    });
  }

  const token = randomBytes(32).toString("hex");
  const expiresAt = new Date(
    Date.now() + SESSION_SECONDS * 1000,
  );

  await db.adminSession.deleteMany({
    where: {
      adminId,
      expiresAt: { lte: new Date() },
    },
  });

  await db.adminSession.create({
    data: {
      adminId,
      tokenHash: tokenHash(token),
      expiresAt,
    },
  });

  cookieStore.set(COOKIE_NAME, token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    path: "/",
    maxAge: SESSION_SECONDS,
    expires: expiresAt,
  });
}

export async function getAdmin() {
  const cookieStore = await cookies();
  const token = cookieStore.get(COOKIE_NAME)?.value;

  if (!token || !/^[a-f0-9]{64}$/.test(token)) {
    return null;
  }

  const session = await db.adminSession.findUnique({
    where: {
      tokenHash: tokenHash(token),
    },
    select: {
      expiresAt: true,
      admin: {
        select: {
          id: true,
          name: true,
          email: true,
          role: true,
          isActive: true,
        },
      },
    },
  });

  if (
    !session ||
    session.expiresAt.getTime() <= Date.now() ||
    !session.admin.isActive
  ) {
    return null;
  }

  return session.admin;
}

export async function requireAdmin() {
  const admin = await getAdmin();

  if (!admin) {
    redirect("/admin/login");
  }

  return admin;
}

export async function deleteAdminSession() {
  const cookieStore = await cookies();
  const token = cookieStore.get(COOKIE_NAME)?.value;

  if (token && /^[a-f0-9]{64}$/.test(token)) {
    await db.adminSession.deleteMany({
      where: {
        tokenHash: tokenHash(token),
      },
    });
  }

  cookieStore.delete(COOKIE_NAME);
}
